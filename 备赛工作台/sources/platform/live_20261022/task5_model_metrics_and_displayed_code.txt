def yolov11(model_size='l', pretrained=True, weights=None): 
    """初始化YOLO模型""" 
    if pretrained and weights is None: 
        if model_size == 'l': 
            model_path = r"yolov11\weights\yolo11l.pt" 
        elif model_size == 's': 
            model_path = r"yolov11\weights\yolo11s.pt" 
        elif model_size == 'm': 
            model_path = r"yolov11\weights\yolo11m.pt" 
        elif model_size == 'n': 
            model_path = r"yolov11\weights\yolo11n.pt" 
        else: 
            raise ValueError("Invalid model size. Choose from 'l', 's', 'm', or 'n'.") 
    else: 
        model_path = weights 
     
    # 初始化模型 
    model = YOLO(model=model_path, task='detect') 
    return model 
 
def train_yolo(args): 
    """使用YOLO内置方法进行训练""" 
    print(f"开始YOLO训练，使用数据增强方案: {args.aug}") 
     
    # 获取数据增强参数 
    aug_params = get_augmentation_params(args.aug) 
     
    # 指定模型文件的完整路径 
    if args.yolo_size == 'l': 
        model_path = r"yolov11\weights\yolo11l.pt" 
    elif args.yolo_size == 's': 
        model_path = r"yolov11\weights\yolo11s.pt" 
    elif args.yolo_size == 'm': 
        model_path = r"yolov11\weights\yolo11m.pt" 
    elif args.yolo_size == 'n': 
        model_path = r"yolov11\weights\yolo11n.pt" 
    else: 
        model_path = r"yolov11\weights\yolo11l.pt" 
     
    data_path = r"yolov11/ultralytics/cfg/datasets/data.yaml" 
     
    model = YOLO(model=model_path, task='detect') 
     
    # 设置训练参数 
    train_kwargs = { 
        'data': data_path, 
        'imgsz': 640, 
        'epochs': args.epochs, 
        'batch': args.batch_size, 
        'workers': args.num_workers, 
        'optimizer': args.optimizer.lower(), 
        'lr0': args.lr, 
        'save_period': 50, 
        'project': r'yolov11\runs\train', 
        'name': f'yolo_{args.yolo_size}_{args.aug}_{args.optimizer.lower()}', 
        'exist_ok': True, 
    } 
     
    # 添加数据增强参数 
    train_kwargs.update(aug_params) 
     
    # 设置设备 
    if torch.cuda.is_available(): 
        train_kwargs['device'] = '0' 
    else: 
        train_kwargs['device'] = 'cpu' 
     
    print(f"YOLO训练参数: {train_kwargs}") 
     
    # 开始训练 
    start_time = time.time() 
    model.train(**train_kwargs) 
    end_time = time.time() 
     
    execution_time = end_time - start_time 
    print(f"YOLO训练完成，执行时间: {execution_time:.2f} 秒") 
     
    # 保存最终权重 
    if hasattr(args, 'weight_name') and args.weight_name: 
        model.save(args.weight_name) 
        print(f"模型已保存到: {args.weight_name}")
