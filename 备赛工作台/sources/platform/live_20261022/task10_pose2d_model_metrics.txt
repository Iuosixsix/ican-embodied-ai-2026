class AugmentationStats: 
    """数据增强统计类""" 
    def __init__(self): 
        self.stats = defaultdict(int) 
        self.total_samples = 0 
     
    def record_augmentation(self, aug_type): 
        self.stats[aug_type] += 1 
        self.total_samples += 1 
     
    def get_stats(self): 
        if self.total_samples == 0: 
            return {} 
        return {k: v/self.total_samples for k, v in self.stats.items()} 
     
    def print_stats(self): 
        stats = self.get_stats() 
        if not stats: 
            return 
        print("数据增强应用统计:") 
        for aug_type, ratio in stats.items(): 
            print(f"  {aug_type}: {ratio:.2%}")

        # Get grasp labels and apply the same transformations

            x = self.numpy_to_torch( 
                np.concatenate( 
                    (np.expand_dims(depth_img, 0),

                ) 
            ) 
        elif self.include_depth:

        elif self.include_rgb: 
            # Normalize RGB after augmentation 
            rgb_img = rgb_img.astype(np.float32)/255.0

            rgb_img = rgb_img.transpose((2, 0, 1)) 
            x = self.numpy_to_torch(rgb_img)

    def apply_augmentation_with_stats(self, img, idx, aug_params): 
        """Apply augmentation strategy with statistics tracking using predefined parameters.""" 
        applied_augs = aug_params['applied_augs'].copy() 
         
        if self.aug_strategy == 'aug0': 
            # Default strategy - no additional augmentation 
            return img, applied_augs 
             
        # Apply geometric transformations that need to be synchronized with labels 
        if aug_params['rotation_angle'] is not None: 
            img = self.apply_rotation_with_angle(img, aug_params['rotation_angle']) 
             
        if aug_params['flip_horizontal']: 
            img = cv2.flip(img, 1) 
             
        if aug_params['flip_vertical']: 
            img = cv2.flip(img, 0) 
             
        if aug_params['scale_factor'] is not None: 
            img = self.apply_scale_crop_with_factor(img, aug_params['scale_factor']) 
             
        if aug_params['translation'] is not None: 
            img = self.apply_translation_with_offset(img, aug_params['translation']) 
             
        if aug_params['occlusion'] is not None: 
            img = self.apply_occlusion_with_params(img, aug_params['occlusion']) 
         
        # Apply color transformations that don't affect labels 
        if self.aug_strategy == 'aug1': 
            img = self.hsv_transform(img) 
             
            if '通道变换' in applied_augs: 
                img = self.channel_shuffle(img) 
                 
        elif self.aug_strategy == 'aug4': 
            if '颜色空间变换' in applied_augs: 
                img = self.color_space_transform(img) 
         
        # Apply complex augmentations that are hard to synchronize with labels 
        # (These are applied after label generation for simplicity) 
        if self.aug_strategy == 'aug3': 
            if 'Mosaic' in applied_augs: 
                img = self.mosaic_augmentation(img, idx) 
                 
            if 'CutMix' in applied_augs: 
                img = self.cutmix_augmentation(img, idx) 
                 
        elif self.aug_strategy == 'aug4': 
            if 'Mixup' in applied_augs: 
                img = self.mixup_augmentation(img, idx) 
         
        # 记录统计信息 
        for aug in applied_augs: 
            self.aug_stats.record_augmentation(aug) 
                 
        return img, applied_augs 
 
    def apply_augmentation_to_grasps(self, bbs, aug_params, img_shape): 
        """将数据增强变换同步应用到抓取框标签上""" 
        h, w = img_shape 
         
        # 应用旋转 
        if aug_params['rotation_angle'] is not None: 
            center = (w // 2, h // 2) 
            bbs.rotate(np.radians(aug_params['rotation_angle']), center) 
         
        # 应用翻转 
        if aug_params['flip_horizontal']: 
            # 水平翻转：x坐标镜像 
            for gr in bbs.grs: 
                gr.points[:, 1] = w - gr.points[:, 1]  # x坐标镜像 
         
        if aug_params['flip_vertical']: 
            # 垂直翻转：y坐标镜像 
            for gr in bbs.grs: 
                gr.points[:, 0] = h - gr.points[:, 0]  # y坐标镜像 
         
        # 应用缩放裁剪 
        if aug_params['scale_factor'] is not None: 
            scale = aug_params['scale_factor'] 
            center = (w // 2, h // 2) 
             
            if scale > 1.0: 
                # 放大后裁剪：需要调整抓取框位置并过滤超出边界的 
                new_size = int(max(h, w) * scale) 
                start = (new_size - max(h, w)) // 2 
                 
                # 先缩放 
                bbs.zoom(scale, center) 
                 
                # 再平移到裁剪后的坐标系 
                bbs.offset((-start, -start)) 
                 
                # 过滤超出边界的抓取框 
                valid_grs = [] 
                for gr in bbs.grs: 
                    if (gr.points[:, 0] >= 0).all() and (gr.points[:, 0] < h).all() and \ 
                       (gr.points[:, 1] >= 0).all() and (gr.points[:, 1] < w).all(): 
                        valid_grs.append(gr) 
                bbs.grs = valid_grs 
            else: 
                # 缩小后填充：直接缩放即可 
                bbs.zoom(scale, center) 
         
        # 应用平移 
        if aug_params['translation'] is not None: 
            dx, dy = aug_params['translation'] 
            bbs.offset((dy, dx))  # 注意：dy对应y轴，dx对应x轴 
         
        # 遮挡不影响抓取框位置，但可能影响可见性 
        # 这里我们保持抓取框不变，让网络学习处理遮挡情况 
         
        return bbs 
 
    def apply_rotation_with_angle(self, img, angle): 
        """使用指定角度应用旋转""" 
        h, w = img.shape[:2] 
        center = (w // 2, h // 2) 
        M = cv2.getRotationMatrix2D(center, angle, 1.0) 
        return cv2.warpAffine(img, M, (w, h), borderMode=cv2.BORDER_REFLECT) 
 
    def apply_scale_crop_with_factor(self, img, scale_factor): 
        """使用指定缩放因子应用缩放裁剪""" 
        h, w = img.shape[:2] 
        scale = scale_factor 
         
        new_h, new_w = int(h * scale), int(w * scale) 
        img = cv2.resize(img, (new_w, new_h)) 
         
        # Random crop back to original size 
        if scale > 1.0: 
            start_h = random.randint(0, new_h - h) 
            start_w = random.randint(0, new_w - w) 
            img = img[start_h:start_h+h, start_w:start_w+w] 
        else:

            pad_h = h - new_h

                                    pad_w//2, pad_w-pad_w//2,  
                                    cv2.BORDER_REFLECT)

    def hsv_transform(self, img): 
        """Apply HSV color space transformation.""" 
        hsv = cv2.cvtColor(img, cv2.COLOR_RGB2HSV) 
        hsv = hsv.astype(np.float32) 
         
        # Random hue shift 
        hue_shift = np.random.uniform(-10, 10) 
        hsv[:, :, 0] = (hsv[:, :, 0] + hue_shift) % 180 
         
        # Random saturation change 
        sat_scale = np.random.uniform(0.8, 1.2) 
        hsv[:, :, 1] = np.clip(hsv[:, :, 1] * sat_scale, 0, 255) 
         
        # Random value change 
        val_scale = np.random.uniform(0.8, 1.2) 
        hsv[:, :, 2] = np.clip(hsv[:, :, 2] * val_scale, 0, 255) 
         
        hsv = hsv.astype(np.uint8) 
        return cv2.cvtColor(hsv, cv2.COLOR_HSV2RGB)

    def mosaic_augmentation(self, img, idx): 
        """Apply Mosaic augmentation (simplified version).""" 
        h, w = img.shape[:2] 
         
        try: 
            indices = [idx] + random.sample(range(len(self.rgb_files)), min(3, len(self.rgb_files)-1)) 
            mosaic = np.zeros((h, w, 3), dtype=np.uint8) 
             
            # Simple 2x2 grid 
            half_h, half_w = h//2, w//2 
            positions = [(0, 0), (0, half_w), (half_h, 0), (half_h, half_w)] 
             
            for i, (pos_idx, (y, x)) in enumerate(zip(indices, positions)): 
                if i < len(indices): 
                    try: 
                        other_img = cv2.imread(self.rgb_files[pos_idx]) 
                        other_img = cv2.resize(other_img, (half_w, half_h)) 
                        mosaic[y:y+half_h, x:x+half_w] = other_img 
                    except: 
                        mosaic[y:y+half_h, x:x+half_w] = img[:half_h, :half_w] 
                else: 
                    mosaic[y:y+half_h, x:x+half_w] = img[:half_h, :half_w] 
                     
            return mosaic 
        except: 
            return img 
 
    def cutmix_augmentation(self, img, idx): 
        """Apply CutMix augmentation.""" 
        try: 
            # Select another random image 
            other_idx = random.randint(0, len(self.rgb_files)-1) 
            other_img = cv2.imread(self.rgb_files[other_idx]) 
            other_img = cv2.resize(other_img, (img.shape[1], img.shape[0])) 
             
            # Random bounding box 
            lam = np.random.beta(1.0, 1.0) 
            cut_ratio = np.sqrt(1. - lam) 
            h, w = img.shape[:2] 
             
            cut_w = int(w * cut_ratio) 
            cut_h = int(h * cut_ratio) 
             
            # Random position 
            cx = np.random.randint(w) 
            cy = np.random.randint(h) 
             
            x1 = np.clip(cx - cut_w // 2, 0, w) 
            y1 = np.clip(cy - cut_h // 2, 0, h) 
            x2 = np.clip(cx + cut_w // 2, 0, w) 
            y2 = np.clip(cy + cut_h // 2, 0, h) 
             
            # Apply CutMix 
            img[y1:y2, x1:x2] = other_img[y1:y2, x1:x2] 
             
        except: 
            pass 
             
        return img 
 
    def mixup_augmentation(self, img, idx): 
        """Apply Mixup augmentation.""" 
        try: 
            # Select another random image 
            other_idx = random.randint(0, len(self.rgb_files)-1) 
            other_img = cv2.imread(self.rgb_files[other_idx]) 
            other_img = cv2.resize(other_img, (img.shape[1], img.shape[0])) 
             
            # Random mixing ratio 
            alpha = 0.2 
            lam = np.random.beta(alpha, alpha) 
             
            # Mix images 
            img = (lam * img + (1 - lam) * other_img).astype(np.uint8) 
             
        except: 
            pass 
             
        return img

def validate(net, device, val_data, batches_per_epoch): 
    """验证函数""" 
    net.eval() 
    results = {'correct': 0, 'failed': 0, 'loss': 0, 'losses': {}} 
    ld = len(val_data) 
    iou_scores = [] 
 
    with torch.no_grad(): 
        batch_idx = 0 
        while batch_idx < batches_per_epoch: 
            for x, y, didx, rot, zoom_factor in val_data: 
                batch_idx += 1 
                if batches_per_epoch is not None and batch_idx >= batches_per_epoch: 
                    break 
 
                xc = x.to(device, non_blocking=True) 
                yc = [yy.to(device, non_blocking=True) for yy in y] 
                lossd = net.compute_loss(xc, yc) 
                loss = lossd['loss'] 
 
                results['loss'] += loss.item()/ld 
                for ln, l in lossd['losses'].items(): 
                    if ln not in results['losses']: 
                        results['losses'][ln] = 0 
                    results['losses'][ln] += l.item()/ld 
 
                q_out, ang_out, w_out = post_process_output(lossd['pred']['pos'], lossd['pred']['cos'], 
                                                            lossd['pred']['sin'], lossd['pred']['width']) 
 
                s = evaluation.calculate_iou_match(q_out, ang_out, 
                                                   val_data.dataset.get_gtbb(didx, rot, zoom_factor), 
                                                   no_grasps=1, 
                                                   grasp_width=w_out) 
                iou_scores.append(s) 
 
                if s: 
                    results['correct'] += 1 
                else: 
                    results['failed'] += 1 
 
    if iou_scores: 
        results['avg_iou'] = np.mean(iou_scores) 
        results['std_iou'] = np.std(iou_scores) 
 
    return results 
 
def train(epoch, net, device, train_data, optimizer, batches_per_epoch, vis=False, use_amp=True, scaler=None): 
    """训练函数""" 
    results = {'loss': 0, 'losses': {}} 
    net.train() 
 
    batch_idx = 0 
    epoch_loss_sum = 0 
    loss_components = defaultdict(list) 
     
    while batch_idx < batches_per_epoch: 
        for x, y, _, _, _ in train_data: 
            batch_idx += 1 
            if batch_idx >= batches_per_epoch: 
                break 
 
            xc = x.to(device, non_blocking=True) 
            yc = [yy.to(device, non_blocking=True) for yy in y] 
 
            optimizer.zero_grad() 
 
            if use_amp and scaler is not None: 
                try: 
                    with autocast(device_type='cuda'): 
                        lossd = net.compute_loss(xc, yc) 
                        loss = lossd['loss'] 
                except TypeError: 
                    with autocast(): 
                        lossd = net.compute_loss(xc, yc) 
                        loss = lossd['loss'] 
 
                scaler.scale(loss).backward() 
                scaler.unscale_(optimizer) 
                torch.nn.utils.clip_grad_norm_(net.parameters(), max_norm=1.0) 
                scaler.step(optimizer) 
                scaler.update() 
            else: 
                lossd = net.compute_loss(xc, yc) 
                loss = lossd['loss'] 
                loss.backward() 
                torch.nn.utils.clip_grad_norm_(net.parameters(), max_norm=1.0) 
                optimizer.step() 
 
            # 记录损失 
            epoch_loss_sum += loss.item() 
            for ln, l in lossd['losses'].items(): 
                loss_components[ln].append(l.item()) 
                if ln not in results['losses']: 
                    results['losses'][ln] = 0 
                results['losses'][ln] += l.item() 
 
            # 日志输出 
            if batch_idx % 100 == 0: 
                current_lr = optimizer.param_groups[0]['lr'] 
                if torch.cuda.is_available(): 
                    memory_used = torch.cuda.memory_allocated(device) / 1024**3 
                    memory_info = f"GPU内存: {memory_used:.2f}GB" 
                else: 
                    memory_info = "CPU模式" 
                 
                logging.info(f'Epoch {epoch+1:02d} | Batch {batch_idx:04d}/{batches_per_epoch} | ' 
                           f'Loss: {loss.item():.4f} | LR: {current_lr:.6f} | {memory_info}') 
 
            # 可视化 
            if vis and batch_idx % 50 == 0: 
                imgs = [] 
                n_img = min(4, x.shape[0]) 
                for idx in range(n_img): 
                    imgs.extend([x[idx,].numpy().squeeze()] + [yi[idx,].numpy().squeeze() for yi in y] + [ 
                        x[idx,].numpy().squeeze()] + [pc[idx,].detach().cpu().numpy().squeeze() for pc in lossd['pred'].values()]) 
                gridshow('Display', imgs, 
                         [(xc.min().item(), xc.max().item()), (0.0, 1.0), (0.0, 1.0), (-1.0, 1.0), (0.0, 1.0)] * 2 * n_img, 
                         [cv2.COLORMAP_BONE] * 10 * n_img, 10) 
                cv2.waitKey(2) 
 
    # 计算平均损失 
    results['loss'] = epoch_loss_sum / batch_idx 
    for l in results['losses']: 
        results['losses'][l] /= batch_idx 
 
    return results

            include_depth=args.use_depth, include_rgb=args.use_rgb, 
            aug_strategy=args.aug, 
            cache_size=config.CACHE_SIZE, enable_cache=config.ENABLE_DISK_CACHE 
        ) 
        print(f" 使用增强版数据集，策略: {args.aug}") 
    elif args.aug == 'fast': 
        # 使用快速数据集 
        train_dataset = FastCornellDataset( 
            args.dataset_path, start=0.0, end=args.split, ds_rotate=args.ds_rotate, 
            random_rotate=True, random_zoom=True, 
            include_depth=args.use_depth, include_rgb=args.use_rgb, 
            aug_strategy=args.aug, 
            enable_cache=config.ENABLE_DISK_CACHE 
        ) 
        print(f" 使用快速数据集，策略: {args.aug}") 
    else: 
        # 使用基础缓存数据集 
        train_dataset = CachedCornellDataset( 
            args.dataset_path, start=0.0, end=args.split, ds_rotate=args.ds_rotate, 
            random_rotate=True, random_zoom=True, 
            include_depth=args.use_depth, include_rgb=args.use_rgb, 
            cache_size=config.CACHE_SIZE, enable_cache=config.ENABLE_DISK_CACHE 
        ) 
        print(f" 使用基础缓存数据集，策略: {args.aug}") 
     
    # 验证集始终使用基础缓存数据集，不使用数据增强 
    val_dataset = CachedCornellDataset( 
        args.dataset_path, start=args.split, end=1.0, ds_rotate=args.ds_rotate, 
        random_rotate=True, random_zoom=True, 
        include_depth=args.use_depth, include_rgb=args.use_rgb, 
        cache_size=config.CACHE_SIZE//2, enable_cache=config.ENABLE_DISK_CACHE 
    ) 
 
    # 优化的DataLoader 
    train_data = torch.utils.data.DataLoader( 
        train_dataset, 
        batch_size=args.batch_size, 
        shuffle=True, 
        num_workers=args.num_workers, 
        pin_memory=config.PIN_MEMORY, 
        persistent_workers=config.PERSISTENT_WORKERS, 
        prefetch_factor=config.PREFETCH_FACTOR, 
        drop_last=True, 
        worker_init_fn=worker_init_fn 
    ) 
     
    val_data = torch.utils.data.DataLoader( 
        val_dataset, 
        batch_size=1, 
        shuffle=False, 
        num_workers=min(args.num_workers, 4), 
        pin_memory=config.PIN_MEMORY, 
        persistent_workers=False, 
        prefetch_factor=2, 
        drop_last=False 
    ) 
 
    print(f" 数据集加载完成: 训练{len(train_dataset)}样本, 验证{len(val_dataset)}样本") 
 
    # 加载模型 
    logging.info(' 正在加载网络模型...') 
    input_channels = 1*args.use_depth + 3*args.use_rgb 
    ggcnn = get_network(args.network) 
    net = ggcnn(input_channels=input_channels) 
    device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu") 
    net = net.to(device) 
 
    # 加载预训练模型 
    if args.pretrained: 
        try: 
            net.load_state_dict(torch.load(args.pretrained, map_location=device)) 
            logging.info(f' 加载预训练模型: {args.pretrained}') 
        except Exception as e: 
            logging.error(f' 加载预训练模型失败: {e}') 
 
    # 配置优化器 
    if args.optimizer == 'sgd': 
        optimizer = optim.SGD(net.parameters(), lr=args.lr, momentum=0.9, weight_decay=args.weight_decay) 
    elif args.optimizer == 'adamw': 
        optimizer = optim.AdamW(net.parameters(), lr=args.lr, weight_decay=args.weight_decay) 
    elif args.optimizer == 'rmsprop': 
        optimizer = optim.RMSprop(net.parameters(), lr=args.lr, alpha=0.99, eps=1e-8, weight_decay=args.weight_decay) 
    else:  # adam 
        optimizer = optim.Adam(net.parameters(), lr=args.lr, weight_decay=args.weight_decay) 
     
    logging.info(f' 使用 {args.optimizer.upper()} 优化器，学习率: {args.lr}') 
 
    # 混合精度训练 
    scaler = None 
    if config.USE_MIXED_PRECISION: 
        try: 
            scaler = GradScaler('cuda') 
        except (TypeError, ValueError): 
            scaler = GradScaler() 
        print(" 启用自动混合精度训练 (AMP)") 
 
    # 打印模型信息 
    print(f"\n  模型参数: {sum(p.numel() for p in net.parameters()):,}") 
    print(f"  计算设备: {device}") 
    if torch.cuda.is_available(): 
        print(f"GPU: {torch.cuda.get_device_name(device)}") 
 
    # 训练循环 
    best_iou = 0.0 
    training_start_time = time.time() 
     
    print(f"\n开始优化训练，共 {args.epochs} 轮") 
    print("="*80) 
     
    for epoch in range(args.epochs): 
        epoch_start_time = time.time() 
         
        print(f"\n Epoch {epoch+1:02d}/{args.epochs}") 
        print("-" * 50) 
         
        # 训练 
        train_results = train(epoch, net, device, train_data, optimizer, args.batches_per_epoch, 
                          vis=args.vis, use_amp=config.USE_MIXED_PRECISION, scaler=scaler) 
         
        # 验证 
        print(f"\n 开始验证...") 
        validation_start = time.time() 
        test_results = validate(net, device, val_data, args.val_batches) 
        validation_time = time.time() - validation_start 
         
        # 计算指标 
        accuracy = test_results['correct'] / (test_results['correct'] + test_results['failed']) 
        iou = test_results.get('avg_iou', accuracy) 
        epoch_time = time.time() - epoch_start_time 
         
        # 记录到TensorBoard 
        tb.add_scalar('loss/train_loss', train_results['loss'], epoch+1) 
        tb.add_scalar('loss/val_loss', test_results['loss'], epoch+1) 
        tb.add_scalar('loss/IOU', accuracy, epoch+1) 
        for n, l in train_results['losses'].items(): 
            tb.add_scalar('train_loss/' + n, l, epoch+1) 
        for n, l in test_results['losses'].items(): 
            tb.add_scalar('val_loss/' + n, l, epoch+1) 
 
        # 打印结果 
        print(f"\n Epoch {epoch+1:02d} 结果:") 
        print(f"   训练损失: {train_results['loss']:.4f}") 
        print(f"   验证损失: {test_results['loss']:.4f}") 
        print(f"   验证准确率: {accuracy:.4f} ({accuracy:.2%})") 
        if 'avg_iou' in test_results: 
            print(f"   平均IoU: {iou:.4f}") 
        print(f"    本轮耗时: {epoch_time:.2f}秒") 
         
        # 每10个epoch打印数据增强统计 
        if (epoch + 1) % 10 == 0 and hasattr(train_dataset, 'aug_stats'): 
            print("\n 数据增强应用统计:") 
            train_dataset.aug_stats.print_stats() 
         
        # 保存模型 
        if accuracy > best_iou or epoch == 0 or ((epoch + 1) % 10) == 0: 
            model_path = os.path.join(save_folder, 'epoch_%02d_iou_%0.2f' % (epoch+1, accuracy)) 
            state_dict_path = os.path.join(save_folder, 'epoch_%02d_iou_%0.2f_statedict.pt' % (epoch+1, accuracy)) 
             
            torch.save(net, model_path) 
            torch.save(net.state_dict(), state_dict_path) 
             
            if accuracy > best_iou: 
                best_iou = accuracy 
                print(f" 新的最佳模型! IoU: {best_iou:.4f}") 
            else: 
                print(f" 定期保存模型: epoch_{epoch+1:02d}_iou_{accuracy:.2f}") 
 
    # 训练完成 
    total_training_time = time.time() - training_start_time 
    print("\n" + "="*80) 
    print(" 优化训练完成!") 
    print("="*80) 
    print(f"  总训练时间: {total_training_time/3600:.2f} 小时") 
    print(f" 最佳验证准确率: {best_iou:.4f} ({best_iou:.2%})") 
    print(f" 模型保存位置: {save_folder}") 
     
    # 打印最终的数据增强统计 
    if hasattr(train_dataset, 'aug_stats'): 
        print("\n 最终数据增强统计:") 
        train_dataset.aug_stats.print_stats() 
     
    print("="*80) 
 
if __name__ == '__main__': 
    logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s') 
    run()
